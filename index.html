// Timer - counts down from 60, persists via timestamp var FM_TS_KEY='fm_ts'; function getRemain(){ var last=parseInt(localStorage.getItem(FM_TS_KEY)||'0'); if(!last){last=Date.now();localStorage.setItem(FM_TS_KEY,last);} var diff=Math.floor((Date.now()-last)/1000); return Math.max(0,60-(diff%60)); } function resetTimer(){localStorage.setItem(FM_TS_KEY,Date.now());} function tickTimer(){ var r=getRemain(); var el=document.getElementById('timerNum'); if(el)el.textContent=r+'s'; if(r<=0){resetTimer();load();} } setInterval(tickTimer,1000); load();setInterval(load,60000);
